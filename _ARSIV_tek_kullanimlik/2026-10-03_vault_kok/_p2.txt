

def _yedek(yol: Path) -> Path:
    damga = datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')
    yedek = yol.with_name(yol.name + '.bak_' + damga)
    shutil.copy2(yol, yedek)
    return yedek
