# ALTYAPI-VDS-GIT-PULL-01 - zamanlanmis git pull (D-364 secenek B)
# Her 5 dk: yalniz bu repoda `git pull --rebase --autostash` (dal chore/monorepo-merge).
# YASAK: docker build/up, force push.
# 2026-10-16 (ihsan): KIRLI-ATLA. MUAF OLMAYAN izlenen dosyada commit edilmemis
#   degisiklik varsa pull YAPILMAZ, tur atlanir, log'a KIRLI-ATLANDI yazilir. Eski
#   davranis (pull --rebase --autostash -> cakisma -> `git rebase --abort`) calisma
#   agacini geri aliyor ve commit edilmemis degisikligi KAYBEDiYORDU. rebase --abort
#   artik yalniz temiz agacta (kayip riski yokken) kurtarma icin kullanilir.
# HARIC listesi ($HARIC): surekli degisen SÜREÇ dosyalari + betigin kendi log'u.
#   Bunlar kirli olsa bile pull'u BLOKLAMAZ (bosuna tur atlanmaz). Baska bir izlenen
#   dosya (orn. agent_bilo/kanit_*.txt) kirliyse pull BLOKLANIR.
# Geri almak icin: schtasks /delete /tn HuginnGitPull /f
$repo  = "C:\Hugin_insights\Apps\huginn-data-insights"
$log   = Join-Path $repo "agent_bilo\kanit_git_pull.txt"
$kilit = Join-Path $env:TEMP "huginn_git_pull.lock"
$GIT   = "C:\Program Files\Git\cmd\git.exe"
$PY    = "C:\Program Files\Python312\python.exe"
$env:Path = "C:\Program Files\Git\cmd;" + $env:Path
$env:HUGINN_AJAN = "bilo"
$env:PYTHONIOENCODING = "utf-8"

# KIRLI kontrolunden MUAF tutulan yollar (surekli degisen surec dosyalari + kendi log).
# Kapsam 2026-10-16 (ihsan): boylece bu dosyalar yuzunden pull her turda atlanmaz.
$HARIC = @(
    ':(exclude)agent_bilo/kanit_git_pull.txt',
    ':(exclude)data/orchestrator/pano_yazim_kaydi.jsonl',
    ':(exclude)data/orchestrator/trigger_log.jsonl',
    ':(glob,exclude)data/orchestrator/triggers/*.jsonl'
)

function Kayit([string]$m) {
    $satir = "{0} {1}" -f (Get-Date -Format "yyyy-MM-dd HH:mm:ss"), $m
    Add-Content -Path $log -Value $satir -Encoding UTF8
    if (Test-Path $log) {
        $s = Get-Content $log -Encoding UTF8 | Select-Object -Last 50   # kanit: son 50 satir
        Set-Content -Path $log -Value $s -Encoding UTF8
    }
}

# ayni anda tek ornek (10 dk icinde ikinci tetiklenme atlanir)
if (Test-Path $kilit) {
    if (((Get-Date) - (Get-Item $kilit).LastWriteTime).TotalMinutes -lt 10) { exit 0 }
}
Set-Content -Path $kilit -Value (Get-Date -Format o)
try {
    Set-Location $repo
    # KIRLI-ATLA: MUAF OLMAYAN izlenen dosyada degisiklik varsa pull YAPMA.
    & $GIT diff --quiet -- . @HARIC 2>$null
    $d1 = $LASTEXITCODE
    & $GIT diff --cached --quiet -- . @HARIC 2>$null
    $d2 = $LASTEXITCODE
    if ($d1 -ne 0 -or $d2 -ne 0) {
        $kirli = (& $GIT status --porcelain -- . @HARIC | Out-String).Trim()
        $kisa = ($kirli -replace '\s+', ' ')
        if ($kisa.Length -gt 160) { $kisa = $kisa.Substring(0, 160) }
        Kayit "KIRLI-ATLANDI | muaf olmayan izlenen dosyada degisiklik var (diff=$d1 cached=$d2) | pull YAPILMADI | $kisa"
        exit 0
    }
    $on = & $GIT rev-parse --short HEAD
    # cmd uzerinden cek: stderr duz metin olarak birlesir (PS ErrorRecord gurultusu olmaz)
    $ham = (cmd /c ('"{0}" pull --rebase --autostash 2>&1' -f $GIT) | Out-String)
    $ciktiCikis = $LASTEXITCODE
    $cikti = (($ham -replace '\s+', ' ')).Trim()
    $cikti = $cikti.Substring(0, [Math]::Min(300, $cikti.Length))
    # DERS (20:36 vakasi): pull exit 0 verip autostash'i uygulayamayip CAKISMALI agac birakabiliyor.
    $u = @(& $GIT ls-files -u); $cozulmemis = $u.Count
    if ($ciktiCikis -eq 0 -and $cozulmemis -eq 0) {
        $son = & $GIT rev-parse --short HEAD
        Kayit "OK $on -> $son | $cikti"
    } else {
        $rebase = (Test-Path (Join-Path $repo ".git\rebase-merge")) -or (Test-Path (Join-Path $repo ".git\rebase-apply"))
        if ($rebase) { & $GIT rebase --abort | Out-Null }
        $neden = if ($ciktiCikis -ne 0) { "exit=$ciktiCikis" } else { "exit=0 ama cozulmemis=$cozulmemis (autostash cakismasi)" }
        Kayit "CAKISMA/NET HATA ($neden; rebase-var=$rebase) | $cikti"
        $ek = if ($rebase) { "rebase abort edildi (agac temiz girdi)" } else { "rebase yok; cozulmemis dosya var, elle cozum gerekiyor (bilo bakar)" }
        $parca = $cikti.Substring(0, [Math]::Min(180, $cikti.Length))
        & $PY (Join-Path $repo "scripts\chat_gonder.py") --to ihsan --type hata --task-id ALTYAPI-VDS-GIT-PULL-01 `
            --mesaj "VDS zamanli git pull CAKISMA ($neden): $parca | $ek | HEAD=$(& $GIT rev-parse --short HEAD). Force YAPILMADI."
    }
} catch {
    Kayit "ISTISNA: $($_.Exception.Message)"
} finally {
    Remove-Item $kilit -Force -ErrorAction SilentlyContinue
}
