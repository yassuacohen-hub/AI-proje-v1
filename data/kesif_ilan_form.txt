		function SepeteEkle(IlanId)
		{
			$.ajax
			(
				{
					type:'POST',
					url:'sepete_ekle.php',
					data:'IlanId=' + IlanId +
						  '&Bos=0',
					success:function(AjaxCevap)
					{
						//Sepet simgesi kaybolsun
						$('#IlanId'+IlanId).html('');
						//Gazete Pdf simgesi kaybolsun
						$('#IlanIdG'+IlanId).html('');
						$('#SpanSepetSayisi').show();
						$('#SpanSepetSayisi').html(AjaxCevap);
					}
				}
			);
		}
		$(document).ready
		(
			function()
			{
				$('.Tarih').on("change",function() { 
				$('#'+$(this).attr('data-name')).toggle(this.checked);
				}).change();
				//İLAN SORGULAMA FORMU
				$('#FormIlanGoruntuleme').submit
				(
					function (e) 
					{
						if (($.trim($('#SicilMudurluguId').val())=='') || ($.trim($('#SicilMudurluguId').val())=='0'))
						{
							toastr.error('Sicil Müdürlüğü Seçiniz','Hata');
							$('#SicilMudurluguId').trigger('chosen:activate');
							return false;
						}
						var Veri;
						Veri = new FormData(this);
						$.ajax
						(
							{
								url: 'ilangoruntuleme_ok.php',
								data: Veri,
								processData: false,
								type: 'POST',
								contentType: false, 
								success: function (AjaxCevap) 
								{
									$('#DivIlanSonuc').html(AjaxCevap);
								}
							}
						);
						e.preventDefault();
					}
				);
				$('#FormIlanGoruntuleme')
				//Tarih
				$('.input-group.date').datepicker
				(
					{
						language: 'TRK'
					}
				);
				$('.DateMasked').mask
				(
					'00.00.0000',
					{
						"placeholder": "gg.aa.yyyy"
					}
				);
			}
		)
